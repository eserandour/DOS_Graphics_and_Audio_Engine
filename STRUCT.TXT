+-- app.h                      flag global quitRequested (Échap → sortie propre)
+-- timer.h / timer.c          timer haute résolution
|                                PIT 8253 reprogrammé à TARGET_HZ (70 Hz)
|                                ISR sur INT 08h chaînée au BIOS à 18,2 Hz
|                                readTimer / elapsedTime / elapsedTimeMs / pause
+-- keyboard.h / keyboard.c    handler clavier ISR sur INT 09h
|                                détecte Échap → quitRequested = 1
|                                toutes les autres touches ignorées
+-- video.h / video.c          mode 13h, double-buffer, synchronisation VGA
|                                SCREEN_WIDTH=320, SCREEN_HEIGHT=200, 1 octet/pixel
|                                backbuffer far alloué par initBackbuffer()
|                                flip() : copie backbuffer → VRAM (0xA000:0000)
|                                waitVRetrace() : attend le retrace vertical (port 0x3DA)
|                                setVideoMode / cursorOn / cursorOff
|                                Macro OFFSET(x,y) = (y<<8)+(y<<6)+x (y*320 sans mul)
|                                Le VGA rafraîchit à 70 Hz ; le rendu est bridé à
|                                ~30 fps par les scènes (render_ms = 33 UL)
|                                flip() ne se synchronise pas sur le vsync
+-- palette.h / palette.c      gestion de la palette DAC VGA 256 couleurs (6 bits/canal)
|                                Palettes globales : defaultPalette, workingPalette,
|                                paletteA, paletteB, grayPalette, pinkPalette,
|                                rainbowPalette
|                                Générateurs : buildGrayPalette / buildPinkPalette
|                                              buildRainbowPalette (cercle HSV 360°)
|                                Manipulation : setPalette / getPalette / copyPalette
|                                               lerpPalette / fadePalette
|                                               cyclePaletteLeft / cyclePaletteRight
|                                savePalette : sauvegarde .pal (768 octets, format DAC natif)
+-- graphics.h / graphics.c    primitives 2D en mode 13h — clipping intégré partout
|                                putPixel / getPixel / clearScreen
|                                drawLine (Cohen-Sutherland + Bresenham entier)
|                                drawRect / drawRectFill (clamp des coordonnées)
|                                drawPolygon / drawPolygonFill (scanline, even-odd rule)
|                                drawCircle / drawCircleFill (mid-point circle, 8 symétries)
+-- image.h / image.c          chargement d'images .raw / .pal
|                                loadImagePal           → workingPalette + DAC
|                                loadImageRaw           → backbuffer (image entière)
|                                loadImageRawKey        → idem avec colorKey
|                                loadScreenRaw          → raccourci 320x200
|                                loadImageZoneRaw       → extraction d'une zone de sprite sheet
|                                loadImageZoneRawKey    → idem avec colorKey
|                                drawImage / drawScreen → palette + pixels en un appel
|                                Codes retour : IMG_OK / IMG_ERR_PAL / IMG_ERR_RAW / IMG_ERR_READ
+-- font1.h / font1.c          moteur texte bitmap multi-tailles
|                                Trois sources : ROM BIOS 8x8, Font1Bank 8x8, 8x16, 16x16
|                                API unifiée via Font1* (BIOS ou Bank selon type)
|                                font1InitBios / font1InitBank8x8 / 8x16 / 16x16
|                                font1DrawChar / font1DrawText / font1DrawTextCentered
|                                font1DefineChar8x8 / 8x16 / 16x16
|                                font1FreeBank : libère la Font1Bank (struct + data far)
|                                Macros CP850 : eAIGU, eGRAVE, eCIRC, aGRAVE, cCED…

+-- scenes/
|   +-- scene0.c    écran noir, 1 s (délimiteur capture vidéo)

|                     fade in 1 s / fade out 3 s (fadePalette non bloquant)
|   +-- scene2.c    démonstration palette VGA, 14 s, 4 phases :
|                     phase 1 (1 s)  : grille 16×16 statique, palette par défaut
|                     phase 2 (5 s)  : cycle droite (cyclePaletteRight)
|                     phase 3 (5 s)  : cycle gauche (cyclePaletteLeft)

|   +-- scene3.c    démonstration polices font1, 6 sous-écrans × 6 s = 36 s :
|                     0 — font1Bios  8x8   (0..127)
|                     1 — font1Bank  8x8   (0..255)
|                     2 — font1Bank  8x16  (0..127)
|                     3 — font1Bank  8x16  (128..255)
|                     4 — font1Bank  16x16 (0..127)
|                     5 — font1Bank  16x16 (128..255)
|                     palette pinkPalette, fade in 2 s / fade out 1 s
|                     libère les Font1Bank avant la fin (heap far pour scene6)
|   +-- scene4.c    affichage HELLO / WORLD avec font2 (32x32), 3 s
|                     "HELLO" centré à y=45, "WORLD" centré à y=122
|   +-- scene5.c    scroller horizontal puis scroller vertical, deux parties enchaînées :
|                     PARTIE A — scroll horizontal (crédits)
|                       police 16X16_F2, défilement droite → gauche
|                       algorithme colonne par colonne (blitColumn)
|                       cadence tick-based : 1 px tous les HSCROLL_TICKS=2 ticks (35 px/s)
|                       deux barres horizontales au-dessus et en-dessous du texte
|                       fin après un passage complet du ruban
|                     TRANSITION A→B
|                       fade-out virgule fixe (fadePaletteInt5, facteur 64→0) en 600 ms
|                     PARTIE B — scroll vertical (générique Star Wars)
|                       lignes de texte défilant bas → haut, fenêtre centrée 240 px
|                       algorithme ligne par ligne (blitVLine), centrage par chaîne
|                       cadence VSCROLL_SPEED_MS=30 ms/pixel
|                       fin quand la dernière ligne quitte le haut de l'écran
|   +-- scene6.c    rotozoom sur freedos.raw (256x256), 10 s
|                     texture découpée en deux blocs far de 32 768 o (tex0, tex1)
|                       (_fmalloc 16 bits : 65 536 déborde → split en 2×32 768)
|                     table sinus 512 entrées en far heap (2 048 o)
|                     arithmétique virgule fixe 16.16 (pas de FPU)
|                     zoom oscillant ×0,6..×2,4, cycle 4 s
|                     phase 1 (0..8 s) : rotozoom ~40 fps, rotation continue
|                     phase 2 (8..10 s) : fade-out palette vers le noir
|   +-- scene7.c    démonstration de toutes les primitives graphics, 18 s, 6 phases × 3 s
|                     phase 1 : tunnel en anneaux (drawCircleFill + drawCircle)
|                     phase 2 : spirale de Galton (drawLine)
|                     phase 3 : plasma en damier morphant (drawRectFill + drawRect)
|                     phase 4 : rosace de polygones (drawPolygon + drawPolygonFill)
|                     phase 5 : rebond de polygones + traînée XOR (getPixel + putPixel)
|                     phase 6 : composition — toutes les primitives simultanées
|                     palette : rainbowPalette, fade in 1 s / fade out 1 s
